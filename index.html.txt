    function normalizePhone(raw) {
      let digits = String(raw || '').replace(/\D/g, '');
      if (digits.startsWith('225') && digits.length === 13) {
        digits = digits.slice(3);
      }
      return digits;
    }

    function formatPhoneDisplay(digits) {
      const clean = normalizePhone(digits);
      if (clean.length === 10) {
        return clean.replace(/(\d{2})(\d{2})(\d{2})(\d{2})(\d{2})/, '$1 $2 $3 $4 $5');
      }
      return digits || '';
    }

    function formatCurrency(amount) {
      return `${(Number(amount) || 0).toLocaleString('fr-FR')} FCFA`;
    }

    function escapeHtml(str) {
      return String(str || '')
        .replace(/&/g, "&amp;")
        .replace(/</g, "&lt;")
        .replace(/>/g, "&gt;")
        .replace(/"/g, "&quot;")
        .replace(/'/g, "&#039;");
    }

    async function saveFile(filename, data) {
      try {
        const dl = window.claude ? await window.claude.use('downloads') : null;
        if (dl) {
          try { await dl.save({ filename, data }); return true; }
          catch (e) { if (e && e.code === 'declined') return false; }
        }
      } catch (e) { /* repli ci-dessous */ }
      const blob = data instanceof Blob ? data : new Blob([data], { type: 'text/csv;charset=utf-8;' });
      const url = URL.createObjectURL(blob);
      const a = document.createElement('a');
      a.href = url; a.download = filename;
      document.body.appendChild(a); a.click(); document.body.removeChild(a);
      setTimeout(() => URL.revokeObjectURL(url), 1000);
      return true;
    }

    function drawRoundedRect(ctx, x, y, w, h, r) {
      ctx.beginPath();
      ctx.moveTo(x + r, y);
      ctx.lineTo(x + w - r, y);
      ctx.quadraticCurveTo(x + w, y, x + w, y + r);
      ctx.lineTo(x + w, y + h - r);
      ctx.quadraticCurveTo(x + w, y + h, x + w - r, y + h);
      ctx.lineTo(x + r, y + h);
      ctx.quadraticCurveTo(x, y + h, x, y + h - r);
      ctx.lineTo(x, y + r);
      ctx.quadraticCurveTo(x, y, x + r, y);
      ctx.closePath();
    }

    function drawRainbowStripOnCanvas(ctx, x, y, width, height) {
      const grad = ctx.createLinearGradient(x, y, x + width, y);
      grad.addColorStop(0.0, "#E63946");
      grad.addColorStop(0.2, "#F77F00");
      grad.addColorStop(0.4, "#FCBF49");
      grad.addColorStop(0.6, "#2A9D8F");
      grad.addColorStop(0.8, "#3A86FF");
      grad.addColorStop(1.0, "#8338EC");
      ctx.fillStyle = grad;
      ctx.fillRect(x, y, width, height);
    }

    function drawEmblemOnCanvas(ctx, cx, cy, radius) {
      ctx.save();
      // Cercle extérieur doré et forêt
      ctx.beginPath();
      ctx.arc(cx, cy, radius, 0, Math.PI * 2);
      ctx.fillStyle = "#1B4332";
      ctx.fill();
      ctx.lineWidth = 3.5;
      ctx.strokeStyle = "#D4AF37";
      ctx.stroke();

      // Cercle intérieur pointillé
      ctx.beginPath();
      ctx.arc(cx, cy, radius - 6, 0, Math.PI * 2);
      ctx.setLineDash([4, 4]);
      ctx.lineWidth = 1.2;
      ctx.strokeStyle = "#F2B705";
      ctx.stroke();
      ctx.setLineDash([]);

      // Croix blanche
      ctx.fillStyle = "#FFFFFF";
      ctx.fillRect(cx - 3.5, cy - 26, 7, 36);
      ctx.fillRect(cx - 16, cy - 15, 32, 6.5);

      // Colombe blanche stylisée
      ctx.beginPath();
      ctx.moveTo(cx - 10, cy - 22);
      ctx.bezierCurveTo(cx - 18, cy - 29, cx - 28, cy - 28, cx - 32, cy - 22);
      ctx.bezierCurveTo(cx - 25, cy - 21, cx - 21, cy - 18, cx - 19, cy - 14);
      ctx.bezierCurveTo(cx - 24, cy - 14, cx - 28, cy - 11, cx - 30, cy - 8);
      ctx.bezierCurveTo(cx - 21, cy - 8, cx - 14, cy - 12, cx - 9, cy - 18);
      ctx.closePath();
      ctx.fillStyle = "#FFFFFF";
      ctx.fill();

      // Bible ouverte dorée
      ctx.fillStyle = "#F2B705";
      ctx.beginPath();
      ctx.moveTo(cx - 26, cy + 14);
      ctx.quadraticCurveTo(cx - 13, cy + 9, cx, cy + 17);
      ctx.quadraticCurveTo(cx + 13, cy + 9, cx + 26, cy + 14);
      ctx.lineTo(cx + 26, cy + 26);
      ctx.quadraticCurveTo(cx + 13, cy + 21, cx, cy + 29);
      ctx.quadraticCurveTo(cx - 13, cy + 21, cx - 26, cy + 26);
      ctx.closePath();
      ctx.fill();

      ctx.strokeStyle = "#0D4A1E";
      ctx.lineWidth = 2;
      ctx.beginPath();
      ctx.moveTo(cx, cy + 17);
      ctx.lineTo(cx, cy + 29);
      ctx.stroke();

      ctx.restore();
    }

    function drawDeterministicSealMatrix(ctx, x, y, size, seedText) {
      ctx.save();
      drawRoundedRect(ctx, x, y, size, size, 10);
      ctx.fillStyle = "#FFFFFF";
      ctx.fill();
      ctx.lineWidth = 2.5;
      ctx.strokeStyle = "#0D4A1E";
      ctx.stroke();

      const cells = 7;
      const pad = 10;
      const cellSize = (size - pad * 2) / cells;
      let hash = 0;
      for (let i = 0; i < seedText.length; i++) {
        hash = ((hash << 5) - hash) + seedText.charCodeAt(i);
        hash |= 0;
      }

      ctx.fillStyle = "#0D4A1E";
      for (let r = 0; r < cells; r++) {
        for (let c = 0; c < cells; c++) {
          const isCorner = (r < 2 && c < 2) || (r < 2 && c >= cells - 2) || (r >= cells - 2 && c < 2);
          const bit = ((Math.abs(hash) >> ((r * cells + c) % 24)) & 1) === 1;
          if (isCorner || bit) {
            ctx.fillRect(x + pad + c * cellSize + 1, y + pad + r * cellSize + 1, cellSize - 2, cellSize - 2);
          }
        }
      }
      ctx.restore();
    }

    function wrapCanvasText(ctx, text, x, y, maxWidth, lineHeight) {
      const words = String(text || '').split(' ');
      let line = '';
      let currentY = y;
      for (let n = 0; n < words.length; n++) {
        const testLine = line + words[n] + ' ';
        const metrics = ctx.measureText(testLine);
        if (metrics.width > maxWidth && n > 0) {
          ctx.fillText(line.trim(), x, currentY);
          line = words[n] + ' ';
          currentY += lineHeight;
        } else {
          line = testLine;
        }
      }
      ctx.fillText(line.trim(), x, currentY);
      return currentY;
    }

    /**
     * Partage l'image PNG du reçu sur WhatsApp via Web Share API (mobile)
     * avec repli automatique par téléchargement PNG + bannière explicative sur ordinateur.
     */
    function shareReceiptPngOnWhatsapp(record) {
      if (!record || (Number(record.amountPaid) || 0) <= 0) return;
      generateReceiptPNG(record);
      const canvas = document.getElementById('receiptGeneratorCanvas');
      const filename = `Recu_Gala_${record.ticketCode || 'JADEM-2026'}.png`;

      canvas.toBlob(async (blob) => {
        if (!blob) {
          downloadReceiptPng(record);
          return;
        }

        const file = new File([blob], filename, { type: 'image/png' });

        // Vérification support partage de fichiers natif (Android / iOS WhatsApp)
        if (navigator.share && navigator.canShare && navigator.canShare({ files: [file] })) {
          try {
            await navigator.share({
              files: [file],
              title: 'Reçu Dîner Gala La Vie Colorée',
              text: 'Voici mon reçu officiel pour le Dîner Gala La Vie Colorée (Temple Emmaüs).'
            });
            showToast("Reçu image PNG partagé !");
            return;
          } catch (err) {
            if (err && err.name === 'AbortError') return;
          }
        }

        // Repli propre sur ordinateur ou navigateur sans partage de fichiers
        if (await saveFile(filename, blob)) showToast("Image PNG téléchargée ! Ouvrez WhatsApp et joignez cette image depuis vos téléchargements.", "info");
      }, 'image/png');
    }

    function renderFinanceAdminDashboard() {
      let totalInscrits = registrations.length;
      let countSoldes = 0;
      let countPartiels = 0;
      let totalEncaisse = 0;
      let resteRecouvrer = 0;

      registrations.forEach(item => {
        const target = Number(item.targetAmount) || 5000;
        const paid = Number(item.amountPaid) || 0;
        const rem = Math.max(0, target - paid);
        totalEncaisse += paid;
        resteRecouvrer += rem;

        if (paid >= target) countSoldes++;
        else if (paid > 0) countPartiels++;
      });

      document.getElementById('kpiTotalInscrits').textContent = totalInscrits;
      document.getElementById('kpiSoldes').textContent = countSoldes;
      document.getElementById('kpiPartiels').textContent = countPartiels;
      document.getElementById('kpiTotalEncaisse').textContent = formatCurrency(totalEncaisse);
      document.getElementById('kpiResteRecouvrer').textContent = formatCurrency(resteRecouvrer);

      const searchQ = document.getElementById('adminSearchInput').value.trim().toLowerCase();
      const filterUnite = document.getElementById('adminFilterUnite').value;
      const filterStatus = document.getElementById('adminFilterStatus').value;

      const filtered = registrations.filter(item => {
        const target = Number(item.targetAmount) || 5000;
        const paid = Number(item.amountPaid) || 0;

        if (filterUnite !== 'ALL' && item.unite !== filterUnite) return false;
        if (filterStatus === 'SOLDE' && paid < target) return false;
        if (filterStatus === 'PARTIEL' && (paid <= 0 || paid >= target)) return false;
        if (filterStatus === 'ATTENTE' && paid > 0) return false;
        if (filterStatus === 'WAVE_DECLARED' && !item.waveDeclarationPending) return false;

        if (searchQ) {
          const hay = `${item.ticketCode} ${item.nom} ${item.prenoms} ${item.phone} ${item.uniteAffichee} ${item.waveRef || ''}`.toLowerCase();
          if (!hay.includes(searchQ)) return false;
        }
        return true;
      });

      if (filtered.length === 0) {
        tbody.innerHTML = `<tr><td colspan="8" class="px-4 py-8 text-center text-gray-500">Aucun participant trouvé pour ce filtre.</td></tr>`;

          : '';

          ? `<button type="button" data-admin-action="preview-png" data-id="${item.id}" class="px-2.5 py-1.5 rounded-lg bg-amber-100 hover:bg-amber-200 text-amber-950 font-bold text-xs transition cursor-pointer">Aperçu Reçu PNG</button>`

    function updateCountdown() {
      const el = document.getElementById('countdownChip');
      const now = new Date(), start = new Date(2027, 0, 1, 0, 0), end = new Date(2027, 0, 1, 5, 50);
      if (now >= end) { el.classList.add('hidden'); return; }
      if (now >= start) { el.textContent = "Le Gala a lieu en ce moment"; return; }
      const d = Math.ceil((start - now) / 864e5);
      el.textContent = d > 1 ? `Plus que ${d} jours avant le Gala` : "C'est cette nuit, dès minuit !";
    }
